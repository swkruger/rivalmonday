// packages/core/src/invite-offer.ts — pure, no imports (client-safe via "@cs/core/invite-offer")
const utcDay = (d: Date): string => d.toISOString().slice(0, 10);

/**
 * Pilot invite links (phase-11-fixes D1): the pilot must still be running after the last day the link can be
 * redeemed. Compares UTC calendar days — `pilotEndsAt` is the end of a UTC day, so a pilot ending on the link's
 * expiry day is refused even though its instant is later. Returns the refusal copy, or null when the dates work.
 */
export function pilotLinkRefusal(pilotEndsAt: Date, linkExpiresAt: Date, now: Date): string | null {
  if (pilotEndsAt.getTime() <= now.getTime()) return 'The pilot end date has already passed. Choose a later date.';
  if (utcDay(pilotEndsAt) <= utcDay(linkExpiresAt)) {
    return `The pilot must end after this link expires on ${utcDay(linkExpiresAt)}. Choose a later pilot end date or a shorter expiry.`;
  }
  return null;
}
```

/** Digits only, US country code dropped: "+1817-286-3655" and "(817) 286-3655" match. Under 7 digits = unusable. */
export const phoneKey = (p: string | null | undefined): string | null => {
  const d = (p ?? '').replace(/\D/g, '');
  const k = d.length === 11 && d.startsWith('1') ? d.slice(1) : d;
  return k.length >= 7 ? k : null;
};
export const domainKey = (d: string | null | undefined): string | null => (d ?? '').trim().toLowerCase().replace(/^www\./, '') || null;

/** D4/P4: one row per Google listing — same place id, or same non-null cid. First (best-ranked) wins; a twin's isClaimed=true carries over. */
export function collapseSameListing<T extends { placeId: string; cid: string | null; isClaimed: boolean | null }>(rows: T[]): T[] {
  const out: T[] = [];
  for (const r of rows) {
    const i = out.findIndex((o) => o.placeId === r.placeId || (r.cid !== null && o.cid === r.cid));
    if (i === -1) out.push(r);
    else if (r.isClaimed === true && out[i]!.isClaimed !== true) out[i] = { ...out[i]!, isClaimed: true };
  }
  return out;
}

/** D4: per row, true when it shares a phone number or website domain with another shown row (Google duplicate listings). */
export function possibleDuplicates(rows: { phone: string | null; domain: string | null }[]): boolean[] {
  const keys = rows.map((r) => {
    const p = phoneKey(r.phone), d = domainKey(r.domain);
    return [p && `p:${p}`, d && `d:${d}`].filter((k): k is string => !!k);
  });
  const seen = new Map<string, number>();
  for (const ks of keys) for (const k of ks) seen.set(k, (seen.get(k) ?? 0) + 1);
  return keys.map((ks) => ks.some((k) => seen.get(k)! > 1));
}
```
